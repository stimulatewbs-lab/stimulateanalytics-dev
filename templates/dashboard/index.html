Code Example (Python/Django)

import requests

url = "https://api.smspro.com/v1/messages"
payload = {
    "to": "+1234567890",
    "message": "Hello from SMS Pro API!",
    "sender_id": "MyCompany"
}
headers = {
    "Authorization": "Bearer YOUR_API_KEY",
    "Content-Type": "application/json"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())
